cmake_minimum_required(VERSION 3.16)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
project(x1000_pet_bridge LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(pet_observer observer.cpp tail.cpp)
add_library(pet_link_wire ../link_wire.c)
add_executable(pet-wire-test ../../../../tools/fixtures/pet_link_wire.c)
target_link_libraries(pet-wire-test PRIVATE pet_link_wire)
if(MSVC)
 target_compile_options(pet_link_wire PUBLIC /utf-8 /W4)
 target_compile_options(pet-wire-test PRIVATE /UNDEBUG)
else()
 target_compile_options(pet_link_wire PUBLIC -Wall -Wextra -Werror)
 target_compile_options(pet-wire-test PRIVATE -UNDEBUG)
endif()
target_include_directories(pet_observer PUBLIC .)
if(MSVC)
 target_compile_options(pet_observer PUBLIC /utf-8 /W4 /Zc:preprocessor)
else()
 target_compile_options(pet_observer PUBLIC -Wall -Wextra -Werror)
endif()
add_executable(pet-inspect inspect.cpp)
target_link_libraries(pet-inspect PRIVATE pet_observer)
add_executable(pet-observer-test ../../../../tools/fixtures/pet_observer.cpp)
target_link_libraries(pet-observer-test PRIVATE pet_observer)
if(MSVC)
 target_compile_options(pet-observer-test PRIVATE /UNDEBUG)
else()
 target_compile_options(pet-observer-test PRIVATE -UNDEBUG)
endif()
enable_testing()
add_test(NAME observer COMMAND pet-observer-test "${CMAKE_BINARY_DIR}/observer-fixtures")
add_test(NAME wire COMMAND pet-wire-test)
get_filename_component(PET_PROJECT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
set(BEARSSL_SOURCE "${PET_PROJECT_ROOT}/work/external-tls-probe/bearssl-0.6" CACHE PATH "Verified BearSSL 0.6 source input")
if(EXISTS "${BEARSSL_SOURCE}/inc/bearssl.h")
 file(GLOB_RECURSE PET_BEARSSL_SOURCES CONFIGURE_DEPENDS "${BEARSSL_SOURCE}/src/*.c")
 add_library(pet_bearssl STATIC ${PET_BEARSSL_SOURCES})
 target_include_directories(pet_bearssl PUBLIC "${BEARSSL_SOURCE}/inc" PRIVATE "${BEARSSL_SOURCE}/src")
 if(MSVC)
  target_compile_options(pet_bearssl PRIVATE /W0)
 endif()
 add_library(pet_discovery_codec ../discovery_wire.c)
 target_link_libraries(pet_discovery_codec PUBLIC pet_bearssl)
 add_executable(pet-discovery-test ../../../../tools/fixtures/pet_discovery_wire.c)
 target_link_libraries(pet-discovery-test PRIVATE pet_discovery_codec)
 if(MSVC)
  target_compile_options(pet-discovery-test PRIVATE /UNDEBUG)
 else()
  target_compile_options(pet-discovery-test PRIVATE -UNDEBUG)
 endif()
 add_test(NAME discovery COMMAND pet-discovery-test)
 find_package(Threads REQUIRED)
 add_executable(pet-bridge server.cpp discovery_server.cpp)
 file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/../VERSION" PET_APP_VERSION LIMIT_COUNT 1)
 target_compile_definitions(pet-bridge PRIVATE PET_BRIDGE_VERSION="${PET_APP_VERSION}")
 target_link_libraries(pet-bridge PRIVATE pet_observer pet_link_wire pet_bearssl pet_discovery_codec Threads::Threads)
 if(WIN32)
  target_link_libraries(pet-bridge PRIVATE ws2_32 bcrypt advapi32)
 endif()
endif()
